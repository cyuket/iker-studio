import type { Faq, Feature } from "./types";

type Def = {
  slug: string;
  name: string;
  h1: string;
  description: string;
  answer: string;
  subline: string;
  /** [kicker label, headline, subline, demo step id]; the kicker number comes from the order. */
  tour: [label: string, headline: string, subline: string, frame: string][];
  points: Feature["points"];
  faqs: Faq[];
  secondary: string[];
  related: string[];
  industries: string[];
  planFeatureKey: string | null;
  flagOnly?: boolean;
};

function feature(d: Def): Feature {
  return {
    slug: d.slug,
    name: d.name,
    title: `${d.name} for Small Business | Iker`,
    description: d.description,
    h1: d.h1,
    answer: d.answer,
    subline: d.subline,
    tour: d.tour.map(([label, headline, subline, frame], i) => ({ kicker: `0${i + 1} · ${label}`, headline, subline, frame })),
    points: d.points,
    faqs: d.faqs,
    keywords: { primary: `${d.name.toLowerCase()} for small business`, secondary: d.secondary },
    related: d.related,
    industries: d.industries,
    planFeatureKey: d.planFeatureKey,
    ...(d.flagOnly ? { flagOnly: true } : {}),
    demo: d.slug,
  };
}
