import type { Faq, Industry } from "./types";

type Def = {
  slug: string;
  name: string; // plural, as used in headings ("Hair Salons")
  noun: string; // lower-case, as used in keywords ("hair salon")
  description: string;
  h1: string;
  answer: string;
  subline: string;
  tour: Industry["tour"];
  pains: string[];
  features: string[];
  depositNorm: string;
  sampleServices: Industry["sampleServices"];
  faqs?: Faq[];
};

const NAIRA = 100; // kobo per naira

function industry(d: Def): Industry {
  const extra: Faq[] = [
    { q: "Do your clients need an account to book?", a: "No. They open your booking link, choose a service and a time, and leave their details." },
    { q: "If I take a deposit, how do clients pay it?", a: "By bank transfer straight to your account. They upload proof and you confirm it in Iker. Deposits are optional per service, and Iker takes no commission." },
  ];
  return {
    slug: d.slug,
    name: d.name,
    noun: d.noun,
    title: `Booking App for ${d.name} | Iker`,
    description: d.description,
    h1: d.h1,
    answer: d.answer,
    subline: d.subline,
    tour: d.tour,
    pains: d.pains,
    features: d.features,
    depositNorm: d.depositNorm,
    sampleServices: d.sampleServices,
    faqs: [...(d.faqs ?? []), ...extra],
    keywords: {
      primary: `${d.noun} booking app`,
      secondary: [
        `appointment booking for ${d.name.toLowerCase()}`,
        `deposit tracking for ${d.name.toLowerCase()}`,
        `${d.noun} booking software Nigeria`,
        `${d.noun} appointment app Nigeria`,
      ],
    },
  };
}

const s = (name: string, minutes: number, naira: number, depositPct: number) => ({
  name,
  minutes,
  priceKobo: naira * NAIRA,
  depositPct,
});
